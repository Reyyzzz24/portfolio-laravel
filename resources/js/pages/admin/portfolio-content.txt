type HeroForm = {

type CollectionKey =

    | 'services'
    | 'projects'
    | 'skills'

type FormErrors = Partial<Record<string, string>>;

const fieldClass =
    'w-full rounded-md border border-input bg-background px-3 py-2 text-sm shadow-xs outline-none transition-[color,box-shadow] focus-visible:border-ring focus-visible:ring-[3px] focus-visible:ring-ring/50';

function fieldError(errors: FormErrors, name: string) {
    return errors[name] ? (
        <p className="mt-1 text-xs text-red-600">{errors[name]}</p>
    ) : null;
}

function TextInput({
    label,
    value,
    onChange,
    error,
    type = 'text',
}: {
    label: string;
    value: string | number | null;
    onChange: (value: string) => void;
    error?: string;
    type?: 'text' | 'number';
}) {
    return (
        <label className="block space-y-1">
            <span className="text-xs font-medium text-muted-foreground">
                {label}
            </span>
            <input
                type={type}
                value={value ?? ''}
                onChange={(event) => onChange(event.target.value)}
                className={fieldClass}
            />
            {error && <p className="text-xs text-red-600">{error}</p>}
        </label>
    );
}

function TextArea({
    label,
    value,
    onChange,
    error,
    rows = 3,
}: {
    label: string;
    value: string | null;
    onChange: (value: string) => void;
    error?: string;
    rows?: number;
}) {
    return (
        <label className="block space-y-1">
            <span className="text-xs font-medium text-muted-foreground">
                {label}
            </span>
            <textarea
                rows={rows}
                value={value ?? ''}
                onChange={(event) => onChange(event.target.value)}
                className={fieldClass}
            />
            {error && <p className="text-xs text-red-600">{error}</p>}
        </label>
    );
}

function ImageUploadInput({
    label,
    value,
    onChange,
    error,
}: {
    label: string;
    value: string | File | null;
    onChange: (file: File) => void;
    error?: string;
}) {
    const [preview, setPreview] = useState<string | null>(
        typeof value === 'string' ? value : null
    );

    return (
        <div className="block space-y-1">
            <span className="text-xs font-medium text-muted-foreground">{label}</span>
            <div className="flex items-center gap-4 rounded-md border p-3 bg-background shadow-xs">
                {preview ? (
                    <img src={preview} alt="Preview" className="h-16 w-28 rounded-md object-cover border" />
                ) : (
                    <div className="flex h-16 w-28 items-center justify-center rounded-md border border-dashed bg-muted text-muted-foreground">
                        <ImagePlus className="h-6 w-6" />
                    </div>
                )}
                <label className="cursor-pointer rounded-md bg-secondary px-3 py-2 text-xs font-medium text-secondary-foreground hover:bg-secondary/80 shadow-xs">
                    Choose File
                    <input
                        type="file"
                        accept="image/*"
                        className="hidden"
                        onChange={(e) => {
                            const file = e.target.files?.[0];
                            if (file) {
                                onChange(file);
                                setPreview(URL.createObjectURL(file));
                            }
                        }}
                    />
                </label>
            </div>
            {error && <p className="text-xs text-red-600">{error}</p>}
        </div>
    );
}

function ColorPickerInput({
    label,
    value,
    onChange,
    error,
}: {
    label: string;
    value: string;
    onChange: (color: string) => void;
    error?: string;
}) {
    // Fungsi pemetaan untuk mengubah text warna dasar menjadi HEX valid
    const formatToHex = (colorVal: string): string => {
        if (!colorVal) return '#3b82f6';
        if (colorVal.startsWith('#')) return colorVal;

        const colorMap: Record<string, string> = {
            blue: '#3b82f6',
            green: '#22c55e',
            purple: '#a855f7',
            red: '#ef4444',
            yellow: '#eab308',
            orange: '#f97316',
            pink: '#ec4899',
            gray: '#6b7280',
            black: '#000000',
            white: '#ffffff'
        };

        return colorMap[colorVal.toLowerCase()] || '#3b82f6';
    };

    const safeHexValue = formatToHex(value);

    return (
        <label className="block space-y-1">
            <span className="text-xs font-medium text-muted-foreground">{label}</span>
            <div className="flex items-center gap-2">
                <input
                    type="color"
                    value={safeHexValue}
                    onChange={(e) => onChange(e.target.value)}
                    className="h-9 w-12 cursor-pointer rounded-md border p-0 bg-background"
                />
                <input
                    type="text"
                    value={value || ''}
                    onChange={(e) => onChange(e.target.value)}
                    className={`${fieldClass} font-mono`}
                    placeholder="#ffffff"
                />
            </div>
            {error && <p className="text-xs text-red-600">{error}</p>}
        </label>
    );
}

function LucideIconPicker({
    label,
    value,
    onChange,
    error,
    color,
}: {
    label: string;
    value: string;
    onChange: (iconName: string) => void;
    error?: string;
    color?: string;
}) {
    const [open, setOpen] = useState(false);
    const iconNames = Object.keys(LucideIcons).filter((key) => {
        const item = (LucideIcons as any)[key];
        return (
            item &&
            /^[A-Z]/.test(key) &&
            (typeof item === 'function' || typeof item === 'object') &&
            !Array.isArray(item)
        );
    });

    const getSafeIconComponent = (name: string) => {
        if (!name) return null;
        const exactName = iconNames.find(i => i.toLowerCase() === name.toLowerCase());
        if (exactName) return (LucideIcons as any)[exactName];
        return null;
    };

    const SelectedIcon = getSafeIconComponent(value);

    return (
        <div className="block space-y-1">
            <span className="text-xs font-medium text-muted-foreground">{label}</span>
            <Popover open={open} onOpenChange={setOpen}>
                <PopoverTrigger asChild>
                    <Button
                        variant="outline"
                        role="combobox"
                        aria-expanded={open}
                        className="w-full justify-between font-normal"
                    >
                        <div className="flex items-center gap-2">
                            {SelectedIcon ? (
                                <SelectedIcon className="h-4 w-4" style={{ color: color || 'currentColor' }} />
                            ) : (
                                <span className="text-muted-foreground text-xs">Select icon...</span>
                            )}
                            <span className="text-sm">{value || "None"}</span>
                        </div>
                    </Button>
                </PopoverTrigger>
                <PopoverContent className="w-[300px] p-0" align="start">
                    <Command>
                        <CommandInput placeholder="Search icon..." />
                        <CommandList>
                            <CommandEmpty>No icon found.</CommandEmpty>
                            <CommandGroup className="max-h-[200px] overflow-y-auto">
                                {iconNames.map((iconName) => {
                                    const IconComponent = getSafeIconComponent(iconName);
                                    if (!IconComponent) return null;
                                    return (
                                        <CommandItem
                                            key={iconName}
                                            value={iconName.toLowerCase()}
                                            onSelect={(currentValue) => {
                                                const originalName = iconNames.find(
                                                    (name) => name.toLowerCase() === currentValue.toLowerCase()
                                                ) || iconName;
                                                onChange(originalName);
                                                setOpen(false);
                                            }}
                                        >
                                            <div className="flex items-center gap-3 w-full">
                                                <IconComponent className="h-4 w-4" style={{ color: color || 'currentColor' }} />
                                                <span>{iconName}</span>
                                            </div>
                                        </CommandItem>
                                    );
                                })}
                            </CommandGroup>
                        </CommandList>
                    </Command>
                </PopoverContent>
            </Popover>
            {error && <p className="text-xs text-red-600">{error}</p>}
        </div>
    );
}

export default function PortfolioContent(props: PortfolioContentForm) {

            hero: props.hero || {

        index: number,
        values: Partial<PortfolioContentForm[Key][number]>,
    ) => {
        const updatedArray = data[key].map((row, rowIndex) =>
            rowIndex === index ? { ...row, ...values } : row,
        );
        setData(key, updatedArray as any);
    };

    // Fungsi global update header section agar semua baris sinkron
    const updateSectionHeader = <Key extends 'services' | 'projects' | 'skills' | 'contacts'>(
        key: Key,
        fields: { section_eyebrow?: string; section_title?: string; section_description?: string }
    ) => {
        const updatedArray = data[key].map((row) => ({
            ...row,
            ...fields
        }));
        setData(key, updatedArray as any);
    };

    // Fungsi Tambah Service Baru
    const addService = () => {
        const first = data.services[0];
        const newService: ServiceForm = {
            id: `new_${Date.now()}`,
            section_eyebrow: first?.section_eyebrow || '',
            section_title: first?.section_title || '',
            title: '',
            description: '',
            color: '#3b82f6',
            icon_path: 'Briefcase',
            display_order: data.services.length + 1,
        };
        setData('services', [...data.services, newService]);
    };

        const first = data.projects[0];

            id: `new_${Date.now()}`,

            section_title: first?.section_title || '',

            image_alt: '',

        const filtered = data[key].filter((_, i) => i !== index);

        event.preventDefault();
        transform((data) => ({
            ...data,
            _method: 'PATCH',

            preserveScroll: true,
            forceFormData: true,
        });
    };

    const educationItems = data.skills.filter(s => s.item_type === 'education');

                            return (
                                <div key={row.id} className="rounded-lg border p-4 bg-background grid grid-cols-1 gap-4 md:grid-cols-2">
                                    <TextInput label="Skill Name" value={row.name} onChange={(v) => updateRow('skills', index, { name: v })} error={errors[`skills.${index}.name`]} />
                                    <TextInput label="Order" type="number" value={row.display_order} onChange={(v) => updateRow('skills', index, { display_order: Number(v) })} error={errors[`skills.${index}.display_order`]} />
                                </div>
                            );
                        })}
                    </CardContent>
                </Card>

                {/* Tampilan Terpisah: Card Khusus EDUCATION */}
                <Card>
                    <CardHeader><CardTitle>Education & History List</CardTitle></CardHeader>
                    <CardContent className="space-y-4">
                        {data.skills.map((row, index) => {
                            if (row.item_type !== 'education') return null;
                            return (
                                <div key={row.id} className="rounded-lg border p-4 bg-background grid grid-cols-1 gap-4 md:grid-cols-2">
                                    <TextInput label="Institution" value={row.institution} onChange={(v) => updateRow('skills', index, { institution: v })} error={errors[`skills.${index}.institution`]} />
                                    <TextInput label="Degree" value={row.degree} onChange={(v) => updateRow('skills', index, { degree: v })} error={errors[`skills.${index}.degree`]} />
                                    <TextInput label="Period" value={row.period} onChange={(v) => updateRow('skills', index, { period: v })} error={errors[`skills.${index}.period`]} />
                                    <TextInput label="Order" type="number" value={row.display_order} onChange={(v) => updateRow('skills', index, { display_order: Number(v) })} error={errors[`skills.${index}.display_order`]} />
                                    <div className="flex items-center gap-2 pt-6">
                                        <input type="checkbox" id={`edu_curr_${index}`} checked={row.is_current} onChange={(e) => updateRow('skills', index, { is_current: e.target.checked })} className="rounded border-gray-300 h-4 w-4" />
                                        <label htmlFor={`edu_curr_${index}`} className="text-sm font-medium">Still Studying Here</label>
                                    </div>
                                </div>
                            );
                        })}
                    </CardContent>
                </Card>

                {/* Contact Section */}
                <Card>
                    <CardHeader><CardTitle>Contact</CardTitle></CardHeader>
                    <CardContent className="space-y-6">
                        <div className="grid grid-cols-1 gap-4 md:grid-cols-2 rounded-lg border border-dashed p-4 bg-muted/10">
                            <TextInput label="Section Eyebrow" value={data.contacts[0]?.section_eyebrow || ''} onChange={(v) => updateSectionHeader('contacts', { section_eyebrow: v })} />
                            <TextInput label="Section Title" value={data.contacts[0]?.section_title || ''} onChange={(v) => updateSectionHeader('contacts', { section_title: v })} />
                        </div>
                        {data.contacts.map((row, index) => (
                            <div key={row.id} className="rounded-lg border p-4 bg-background grid grid-cols-1 gap-4 md:grid-cols-2">
                                <TextInput label="Form Action" value={row.form_action} onChange={(v) => updateRow('contacts', index, { form_action: v })} error={errors[`contacts.${index}.form_action`]} />
                                <TextInput label="Submit Label" value={row.submit_label} onChange={(v) => updateRow('contacts', index, { submit_label: v })} error={errors[`contacts.${index}.submit_label`]} />
                                <TextInput label="Label" value={row.label} onChange={(v) => updateRow('contacts', index, { label: v })} error={errors[`contacts.${index}.label`]} />
                                <TextInput label="Href" value={row.href} onChange={(v) => updateRow('contacts', index, { href: v })} error={errors[`contacts.${index}.href`]} />
                                <LucideIconPicker label="Contact Icon" value={row.icon_path} onChange={(v) => updateRow('contacts', index, { icon_path: v })} error={errors[`contacts.${index}.icon_path`]} />
                                <TextInput label="Order" type="number" value={row.display_order} onChange={(v) => updateRow('contacts', index, { display_order: Number(v) })} error={errors[`contacts.${index}.display_order`]} />
                                <div className="md:col-span-2"><TextArea label="Value" value={row.value} onChange={(v) => updateRow('contacts', index, { value: v })} error={errors[`contacts.${index}.value`]} /></div>
                            </div>
                        ))}
                    </CardContent>
                </Card>

                {/* Footer Section */}
                <EditableRows title="Footer" rows={data.footers} errors={errors} prefix="footers" render={(row, index) => (
                    <>
                        <TextInput label="Copyright" value={row.copyright} onChange={(v) => updateRow('footers', index, { copyright: v })} error={errors[`footers.${index}.copyright`]} />
                        <TextInput label="Name" value={row.name} onChange={(v) => updateRow('footers', index, { name: v })} error={errors[`footers.${index}.name`]} />
                        <TextInput label="URL" value={row.url} onChange={(v) => updateRow('footers', index, { url: v })} error={errors[`footers.${index}.url`]} />
                        <TextInput label="Hover Class" value={row.hover_class} onChange={(v) => updateRow('footers', index, { hover_class: v })} error={errors[`footers.${index}.hover_class`]} />
                        <TextInput label="Order" type="number" value={row.display_order} onChange={(v) => updateRow('footers', index, { display_order: Number(v) })} error={errors[`footers.${index}.display_order`]} />
                        <div className="md:col-span-2"><LucideIconPicker label="Footer Icon" value={row.icon_path} onChange={(v) => updateRow('footers', index, { icon_path: v })} error={errors[`footers.${index}.icon_path`]} /></div>
                    </>
                )} />
            </form>
        </>
    );
}

function EditableRows<Row extends { id: number | string;[key: string]: any }>({
    title, rows, errors, prefix, render,
}: {
    title: string; rows: Row[]; errors: FormErrors; prefix: string; render: (row: Row, index: number) => ReactNode;
}) {
    return (
        <Card>
            <CardHeader><CardTitle>{title}</CardTitle></CardHeader>
            <CardContent className="space-y-4">
                {rows.map((row, index) => (
                    <div key={row.id} className="rounded-lg border bg-muted/20 p-4">
                        <div className="mb-4 flex items-center justify-between gap-3">
                            <h3 className="font-medium">{row.title || row.label || row.name || `${title} ${index + 1}`}</h3>
                            <span className="text-xs text-muted-foreground">#{row.id}</span>
                        </div>
                        <div className="grid grid-cols-1 gap-4 md:grid-cols-2">{render(row, index)}</div>
                        {fieldError(errors, `${prefix}.${index}.id`)}
                    </div>
                ))}
            </CardContent>
        </Card>
    );
}
